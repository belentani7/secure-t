<script>
const DATA_URL = './data/threat-intel.json';
const PAGE_SIZE = 25;

let allKev = [];
let filteredKev = [];
let currentPage = 0;
let activeFilter = 'all';
let searchTerm = '';
let stats = {};
let nvdItems = [];

async function loadData() {
  try {
    const r = await fetch(DATA_URL);
    if (!r.ok) throw new Error(`HTTP ${r.status}`);
    const data = await r.json();

    stats = data.stats || {};
    allKev = data.kev?.recent || [];
    nvdItems = data.nvd_critical_recent || [];

    const generated = Date.parse(data.generated_utc);
    if (!Number.isFinite(generated) || generated > Date.now() + 86400000) throw new Error('Fecha de generación incorrecta');
    const ageDays = Math.max(0, Math.floor((Date.now() - generated) / 86400000));
    document.getElementById('stamp').textContent =
      'Generado: ' + new Date(generated).toLocaleString('es-ES', {dateStyle:'medium',timeStyle:'short',timeZone:'UTC'}) +
      ' UTC · Catálogo CISA ' + String(data.kev?.catalog_version || '—');
    document.getElementById('freshness').textContent =
      ageDays > 2
        ? '⚠ Atención: esta instantánea tiene ' + ageDays + ' días. NO refleja nuevas alertas; revisa CISA/NVD oficiales.'
        : 'Instantánea reciente (' + ageDays + ' días). Aun así, no hay consulta en vivo a CISA ni NVD.';

    document.getElementById('s-total').textContent = (stats.total_kev || 0).toLocaleString('es');
    document.getElementById('s-ransomware').textContent = (stats.ransomware_linked || 0).toLocaleString('es');
    document.getElementById('s-pct').textContent = `${stats.ransomware_pct || 0}%`;
    document.getElementById('s-nvd').textContent = nvdItems.length;

    applyFilter();
    renderNvd();
    renderStats();
  } catch(e) {
    const area = document.getElementById('kev-table-wrap');
    area.replaceChildren();
    const message = document.createElement('div');
    message.className = 'err-msg';
    message.textContent = '⚠ No se pudo cargar el dataset. Consulta los enlaces oficiales y comprueba la publicación.';
    area.append(message);
    document.getElementById('freshness').textContent = '⚠ Datos locales no disponibles.';
  }
}

function applyFilter() {
  const q = searchTerm.toLowerCase();
  filteredKev = allKev.filter(v => {
    const matchFilter = activeFilter === 'all' || (activeFilter === 'ransomware' && v.ransomware);
    const matchSearch = !q ||
      v.cve_id?.toLowerCase().includes(q) ||
      v.vendor?.toLowerCase().includes(q) ||
      v.product?.toLowerCase().includes(q) ||
      v.name?.toLowerCase().includes(q) ||
      v.short_description?.toLowerCase().includes(q);
    return matchFilter && matchSearch;
  });
  currentPage = 0;
  renderKevTable();
}

function renderKevTable() {
  const start = currentPage * PAGE_SIZE;
  const slice = filteredKev.slice(start, start + PAGE_SIZE);
  const total = filteredKev.length;

  if (!slice.length) {
    document.getElementById('kev-table-wrap').innerHTML = '<div class="loading-msg">Sin resultados.</div>';
    document.getElementById('kev-pager').style.display = 'none';
    return;
  }

  const today = new Date().toISOString().slice(0,10);
  const isNew = d => d && (new Date(today) - new Date(d)) < 14*24*60*60*1000;

  let html = '<div class="table-wrap"><table><thead><tr><th>CVE</th><th>Producto</th><th>Añadida</th><th>Estado</th><th>Descripción</th></tr></thead><tbody>';
  for (const v of slice) {
    const badges = [];
    if (v.ransomware) badges.push('<span class="badge badge-ransomware">ransomware</span>');
    if (isNew(v.date_added)) badges.push('<span class="badge badge-new">nueva</span>');
    html += `<tr>
      <td><span class="cve-id">${esc(v.cve_id)}</span></td>
      <td><span class="product-name">${esc(v.product)}</span><br><span class="vendor">${esc(v.vendor)}</span></td>
      <td><span class="date">${esc(v.date_added || '—')}</span></td>
      <td>${badges.join(' ')}</td>
      <td><span class="desc">${esc(v.short_description || v.name || '')}</span></td>
    </tr>`;
  }
  html += '</tbody></table></div>';

  document.getElementById('kev-table-wrap').innerHTML = html;

  const totalPages = Math.ceil(total / PAGE_SIZE);
  document.getElementById('kev-pager').style.display = total > PAGE_SIZE ? 'flex' : 'none';
  document.getElementById('p-info').textContent = `Página ${currentPage+1} / ${totalPages} — ${total} resultados`;
  document.getElementById('p-prev').disabled = currentPage === 0;
  document.getElementById('p-next').disabled = currentPage >= totalPages - 1;
}

function renderNvd() {
  if (!nvdItems.length) {
    document.getElementById('nvd-list').innerHTML = '<div class="loading-msg">Sin datos NVD.</div>';
    return;
  }
  let html = '';
  for (const v of nvdItems) {
    html += `<div class="nvd-item">
      <div class="nvd-header">
        <span class="nvd-cve">${esc(v.cve_id)}</span>
        <span class="nvd-score">CVSS ${esc(v.score)}</span>
        <span class="nvd-date">${esc(v.published)}</span>
        ${v.attack_vector ? `<span class="badge badge-critical">${esc(v.attack_vector)}</span>` : ''}
      </div>
      <p class="nvd-desc">${esc(v.description)}</p>
    </div>`;
  }
  document.getElementById('nvd-list').innerHTML = html;
}

function renderStats() {
  // Vendor bar chart
  const vendors = stats.top_vendors || [];
  const maxV = Math.max(1, ...vendors.map(v => safeNumber(v.count)));
  document.getElementById('vendor-chart').innerHTML = vendors.map(v => `
    <div class="bar-row">
      <span class="bar-label" title="${esc(v.vendor)}">${esc(v.vendor)}</span>
      <div class="bar-track"><div class="bar-fill" style="width:${Math.max(0,Math.min(100,Math.round(safeNumber(v.count)/maxV*100)))}%"></div></div>
      <span class="bar-count">${safeNumber(v.count)}</span>
    </div>`).join('');

  // CWE list
  const cwes = stats.top_cwes || [];
  document.getElementById('cwe-list').innerHTML = cwes.map(c => `
    <div class="cwe-item"><span class="cwe-id">${esc(c.cwe)}</span><span class="cwe-count">${safeNumber(c.count)}</span></div>`).join('');

  // Timeline SVG
  const tl = stats.timeline || [];
  if (!tl.length) return;
  const W = 780, H = 120, PL = 40, PR = 10, PT = 10, PB = 30;
  const cw = W - PL - PR, ch = H - PT - PB;
  const maxC = Math.max(1, ...tl.map(t => safeNumber(t.count)));
  const bw = cw / tl.length;

  let svgBars = '', svgLabels = '';
  tl.forEach((t, i) => {
    const bh = (safeNumber(t.count) / maxC) * ch;
    const x = PL + i * bw + bw * .1;
    const y = PT + ch - bh;
    svgBars += `<rect x="${x.toFixed(1)}" y="${y.toFixed(1)}" width="${(bw*.8).toFixed(1)}" height="${bh.toFixed(1)}" fill="var(--accent)" opacity=".75" rx="2"><title>${esc(t.month)}: ${safeNumber(t.count)}</title></rect>`;
    if (i % 4 === 0 || i === tl.length - 1) {
      svgLabels += `<text x="${(x + bw*.4).toFixed(1)}" y="${H - 5}" text-anchor="middle" fill="var(--text-3)" font-size="9" font-family="monospace">${esc(String(t.month || "").slice(2))}</text>`;
    }
  });
  document.getElementById('timeline-svg').innerHTML = `<svg viewBox="0 0 ${W} ${H}" xmlns="http://www.w3.org/2000/svg" style="width:100%;height:auto">${svgBars}${svgLabels}</svg>`;
}

// Tab switching
document.querySelectorAll('.tab').forEach(btn => {
  btn.addEventListener('click', () => {
    document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
    btn.classList.add('active');
    ['kev','nvd','stats'].forEach(id => {
      document.getElementById('tab-' + id).classList.toggle('hidden', id !== btn.dataset.tab);
    });
  });
});

// Search
document.getElementById('kev-search').addEventListener('input', e => {
  searchTerm = e.target.value;
  applyFilter();
});

// Filters
document.querySelectorAll('[data-filter]').forEach(btn => {
  btn.addEventListener('click', () => {
    activeFilter = btn.dataset.filter;
    document.querySelectorAll('[data-filter]').forEach(b => b.classList.remove('active'));
    btn.classList.add('active');
    applyFilter();
  });
});

// Pagination
document.getElementById('p-prev').addEventListener('click', () => { currentPage--; renderKevTable(); window.scrollTo({top:0,behavior:'smooth'}); });
document.getElementById('p-next').addEventListener('click', () => { currentPage++; renderKevTable(); window.scrollTo({top:0,behavior:'smooth'}); });

function safeNumber(value) {
  const n = Number(value);
  return Number.isFinite(n) && n >= 0 ? Math.min(n, 1000000000) : 0;
}
function esc(value) {
  return String(value ?? '').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;').replace(/'/g,'&#39;');
}

loadData();
</script>
