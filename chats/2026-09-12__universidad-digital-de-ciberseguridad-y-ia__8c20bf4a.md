def _generate_seo_artifacts(self, domain: str = "https://secure-t.local"):
    """Genera sitemap.xml y robots.txt reales basados en el HTML generado."""
    public_dir = self.root_dir / "public"
    public_dir.mkdir(exist_ok=True)
    
    # 1. Sitemap dinámico
    urls = [f"{domain}/", f"{domain}/campus/"]
    campus_dir = self.root_dir / "campus"
    if campus_dir.exists():
        for html_file in campus_dir.rglob("*.html"):
            rel_path = html_file.relative_to(self.root_dir).as_posix()
            urls.append(f"{domain}/{rel_path}")
            
    xml = ['<?xml version="1.0" encoding="UTF-8"?>', '<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">']
    for u in urls:
        xml.extend([f"  <url>", f"    <loc>{u}</loc>", f"    <changefreq>monthly</changefreq>", f"  </url>"])
    xml.append("</urlset>")
    (public_dir / "sitemap.xml").write_text("\n".join(xml), encoding="utf-8")
    
    # 2. Robots.txt permisivo para el campus real
    robots = f"""User-agent: *
Allow: /
Allow: /campus/
Allow: /campus/cursos/
Disallow: /api/
Disallow: /admin/

Sitemap: {domain}/public/sitemap.xml
"""
    (public_dir / "robots.txt").write_text(robots, encoding="utf-8")
